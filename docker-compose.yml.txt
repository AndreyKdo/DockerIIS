  # Sitio web (IIS) - contenedor Windows

  web:
    build:
      context: ./web
    container_name: web-iis
    ports:
      - "8000:80"
    volumes:
      - ./web/wwwroot:C:/inetpub/wwwroot
    restart: unless-stopped
    networks:

      context: ./api-pagos

    networks:
      - dockerlab

    container_name: api-consultas
